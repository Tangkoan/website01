3. នេះគឺសម្រាប់ការ setup ភាសាអោយ Laravel Backend របស់យើង

> php artisan lang:publish

បង្កើត Middleware ដើម្បីចងចាំភាសា
> php artisan make:middleware SetLocale

បន្ទាប់មក បើកឯកសារ app/Http/Middleware/SetLocale.php រួចកែសម្រួលកូដដូចខាងក្រោម៖
        <?php
        namespace App\Http\Middleware;
        use Closure;
        use Illuminate\Http\Request;
        use Illuminate\Support\Facades\App;
        use Illuminate\Support\Facades\Session;
        class SetLocale
        {
            public function handle(Request $request, Closure $next)
            {
                // បើមានការកំណត់ភាសាក្នុង Session, យកវាមកប្រើ។ បើគ្មានទេ យក 'km' ជាភាសាដើម
                if (Session::has('locale')) {
                    App::setLocale(Session::get('locale'));
                } else {
                    App::setLocale('km'); 
                }

                return $next($request);
            }
        }


កំណត់ Middleware និងបង្កើត Route ប្តូរភាសា
    ទី១៖ ចាប់ពី Laravel 11/12 មក យើងត្រូវចុះឈ្មោះ Middleware នៅក្នុងឯកសារ bootstrap/app.php។ សូមបើកវា រួចបន្ថែម Middleware ចូល៖

        ->withMiddleware(function (Middleware $middleware) {
            // បន្ថែមបន្ទាត់នេះចូល
            $middleware->web(append: [
                \App\Http\Middleware\SetLocale::class,
            ]);
        })



ទី២៖ បើកឯកសារ routes/web.php រួចបន្ថែម Route មួយនេះសម្រាប់ពេលគេចុចប្តូរភាសា៖// កូដចាស់របស់អ្នកនៅរក្សាទុកដដែល...
        // Route សម្រាប់ប្តូរភាសា
        Route::get('/lang/{locale}', function ($locale) {
            // កំណត់ឱ្យប្រើបានតែ en និង km ប៉ុណ្ណោះ
            if (in_array($locale, ['en', 'km'])) {
                session()->put('locale', $locale);
            }
            return back(); // រុញត្រឡប់ទៅទំព័រដើមវិញ
        })->name('switch.language');
